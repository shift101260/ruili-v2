// Google Apps Script 後端程式碼 function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); // 依據您的 Excel 欄位架構組合資料（共 14 個欄位） var newRow = [ data.caseNo || "", // 1. 案件編號 data.date || "", // 2. 建檔日期 data.mainCategory || "", // 3. 主分類 data.subStatus || "", // 4. 細分狀態 data.clientName || "", // 5. 客戶名稱 data.location || "", // 6. 標的地址/地號 data.quotedAmount || 0, // 7. 報價金額 data.signedAmount || 0, // 8. 簽約金額 data.solarArea || "-", // 9. 太陽能-可建置面積(坪) data.solarCapacity || "-", // 10. 太陽能-預估設置容量(kWp) data.solarPower || "-", // 11. 太陽能-預估年發電量(度) data.solarRevenue || "-", // 12. 太陽能-預估首年躉售收益(元) data.notes || "", // 13. 案件進度與備註 data.nasPath || "" // 14. NAS 資料夾路徑 ]; // 將新資料自動寫入試算表最後一列 sheet.appendRow(newRow); // 回傳成功訊息 return ContentService.createTextOutput(JSON.stringify({ "status": "success", "message": "案件已成功同步至 Excel！" })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": error.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // 供系統前端讀取所有案件資料用 (GET) function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var rows = sheet.getDataRange().getValues(); return ContentService.createTextOutput(JSON.stringify({ "status": "success", "data": rows })) .setMimeType(ContentService.MimeType.JSON); }
